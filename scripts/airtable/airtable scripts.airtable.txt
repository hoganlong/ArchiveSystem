


  // === CONFIGURE THESE ===
  const TABLE_NAME = 'ARTWORK';
  const SOURCE_FIELD = 'PAPER_INVENTORY';
  const TARGET_FIELD = 'Polaroid';
  const PREFIX = 'Polaroid';
  const SEPARATOR = ' - ';
  const DRY_RUN = false;   // <-- set to false to actually write
  // =======================

  const table = base.getTable(TABLE_NAME);
  const query = await table.selectRecordsAsync({ fields: [SOURCE_FIELD, TARGET_FIELD] });

  const updates = [];
  const preview = [];
  for (const record of query.records) {
      const fileName = record.getCellValueAsString(SOURCE_FIELD);
      if (!fileName.startsWith(PREFIX)) continue;

      const idx = fileName.indexOf(SEPARATOR);
      if (idx === -1) continue;

      const extracted = fileName.substring(idx + SEPARATOR.length);
      const currentTarget = record.getCellValueAsString(TARGET_FIELD);

      updates.push({ id: record.id, fields: { [TARGET_FIELD]: extracted } });
      preview.push({
          recordId: record.id,
          source: fileName,
          currentTarget: currentTarget,
          newTarget: extracted,
          changes: currentTarget !== extracted ? 'YES' : 'no',
      });
  }

  output.text(`Found ${preview.length} matching record(s).`);
  output.table(preview);

  if (DRY_RUN) {
      output.text('DRY RUN — no changes written. Set DRY_RUN = false to apply.');
  } else {
      while (updates.length > 0) {
          await table.updateRecordsAsync(updates.slice(0, 50));
          updates.splice(0, 50);
      }
      output.text('Done.');
  }
